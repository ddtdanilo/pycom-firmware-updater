    def __skipBoardPage(self):
        if self.__localFileFlash:
            self.__downloadWaitLabel.setText(('Please wait while preparing firmware file <br><br>{}').format(humanfriendly.format_path(self.__localFileName)))
            self.__downloadWaitPage.setTitle('Reading local file')
            self.__set_firmware_wait_label(None)
            self.__firmwareVersionLabel2.setText('Updating from file: <b>' + self.__localFileName.split('/')[-1] + '</b>')
            if os.path.isfile(self.__localFileName):
                if self.__forceNewLoRaRegionSelector.isChecked():
                    return CustomWizard.COUNTRY_IDX
                else:
                    return self.__check_advanced_page(CustomWizard.DOWNLOAD_IDX)

            else:
                self.__finishedUpgrade(2)
                return CustomWizard.RESULT_IDX
        else:
            self.__downloadWaitLabel.setText('Please wait, firmware download in progress...')
            self.__downloadWaitPage.setTitle('Download')
            self.__firmwareVersionLabel2.setText('')
        if self.__res_json:
            if self.check_lora_region(self.__res_json['firmware_type'], self.__lora_region):
                print_debug('We have check_lora_region=True')
                print_debug(('self.__firmwareTypeSelector.currentText()={}').format(self.__firmwareTypeSelector.currentText()))
                print_debug(('self.__forceupdate_pybytes? {}').format(self.__forceupdate_pybytes))
                print_debug(('self.check_pybytes_settings(self.__pybytes_device_token)? = {}').format(self.check_pybytes_settings(self.__pybytes_device_token)))
                print_debug(('str(self.__firmwareTypeSelector.currentText) in self.__pybytes_firmwares? {}').format(str(self.__firmwareTypeSelector.currentText) in self.__pybytes_firmwares))
                print_debug(('self.__forceupdate_pybytes or not self.check_pybytes_settings(self.__pybytes_device_token)? {}').format(self.__forceupdate_pybytes or not self.check_pybytes_settings(self.__pybytes_device_token)))
                print_debug(('str(self.__firmwareTypeSelector.currentText) in self.__pybytes_firmwares and (self.__forceupdate_pybytes or not self.check_pybytes_settings(self.__pybytes_device_token))? {}').format(str(self.__firmwareTypeSelector.currentText) in self.__pybytes_firmwares and (self.__forceupdate_pybytes or not self.check_pybytes_settings(self.__pybytes_device_token))))
                if self.page(CustomWizard.BOARD_IDX).field('board_selector').toInt()[0] == 5 or self.page(CustomWizard.BOARD_IDX).field('board_selector').toInt()[0] == 6 or self.__checkSigfoxCredentials():
                    return CustomWizard.REGION_IDX
                if not self.__enablePybytesRegSelector.isChecked() and str(self.__firmwareTypeSelector.currentText()) in self.__pybytes_firmwares or str(self.__firmwareTypeSelector.currentText()) in self.__pybytes_firmwares and (self.__forceupdate_pybytes or not self.check_pybytes_settings(self.__pybytes_device_token)):
                    return CustomWizard.PYBYTES_IDX
                if not self.__info_obtained:
                    print_debug('Return self.__check_advanced_page(WAIT_IDX) from __skipBoardPage')
                    return self.__check_advanced_page(CustomWizard.WAIT_IDX)
                print_debug('Return self.__check_advanced_page(DOWNLOAD_IDX) from __skipBoardPage')
                return self.__check_advanced_page(CustomWizard.DOWNLOAD_IDX)
            else:
                print_debug('Return COUNTRY_IDX from __skipBoardPage')
                return CustomWizard.COUNTRY_IDX
        else:
            print_debug('We do not have self.__res_json, BOARD_IDX from __skipBoardPage')
            if self.__wizard_retry:
                return CustomWizard.SERIAL_IDX
            return CustomWizard.BOARD_IDX
        return

    def __skipPybytesPage(self):
        print_debug('-->-->-->-->-->-->-->-->-->-->-->-->    This is __skipPybytesPage   <--<--<--<--<--<--<--<--<--<--<--<--<--<--<--<--<--<--<--')
        print_debug('self.__info_obtained? %s' % self.__info_obtained)
        print_debug('self.__pybytes_info_obtained: %s' % str(self.__pybytes_info_obtained))
        print_debug('Firmware Type: ' + self.__firmwareTypeSelector.currentText())
        print_debug(('Board COnnection: {}').format(self.__board_connection_failed))
        print_debug(('str(self.__firmwareTypeSelector.currentText()) not in self.__pybytes_firmwares = {}').format(str(self.__firmwareTypeSelector.currentText()) not in self.__pybytes_firmwares))
        print_debug(('str(self.__firmwareTypeSelector.currentText()) in self.__pybytes_firmwares = {}').format(str(self.__firmwareTypeSelector.currentText()) in self.__pybytes_firmwares))
        if self.__board_connection_failed:
            self.__finishedUpgrade(1)
            return CustomWizard.RESULT_IDX
        else:
            if self.__localFileFlash or self.__skip_pybytes or self.__pybytes_info_obtained or str(self.__firmwareTypeSelector.currentText()) not in self.__pybytes_firmwares or self.check_pybytes_settings(self.__pybytes_device_token) and not self.__forceNewPybytesRegSelector.isChecked():
                print_debug('Return self.__check_advanced_page(DOWNLOAD_IDX) from __skipPybytesPage')
                return self.__check_advanced_page(CustomWizard.DOWNLOAD_IDX)
            if not self.__enablePybytesRegSelector.isChecked() and str(self.__firmwareTypeSelector.currentText()) in self.__pybytes_firmwares:
                print_debug('Return self.__check_advanced_page(DOWNLOAD_IDX) from __skipPybytesPage')
                return self.__check_advanced_page(CustomWizard.DOWNLOAD_IDX)
            if not self.__info_obtained:
                self.__fromPybytes = True
                self.__insert_device()
            elif self.__checkSigfoxCredentials():
                self.__fromPybytes = True
                print_debug('Invoking self.__register_sigfox from __skipPybytesPage')
                self.__register_sigfox()
            wizard.button(QtGui.QWizard.CustomButton1).setText('Skip')
            print_debug('Returning PYBYTES_IDX from __skipPybytesPage (set button to Skip!)')
            return CustomWizard.PYBYTES_IDX

        return

    def __newInfoTypeSelected(self):
        if self.__res_json:
            config_block = base64.b64decode(self.__res_json['binary'])
        else:
            config_block = self.__config_block
        try:
            sigfox_id = str(binascii.hexlify(config_block[8:12]).upper())
            sigfox_pac = str(binascii.hexlify(config_block[12:20]).upper())
        except:
            sigfox_id = 'FFFFFFFF'

        try:
            self.__loramac_s = '%02X%02X%02X%02X%02X%02X%02X%02X' % struct.unpack('BBBBBBBB', config_block[:8])
        except:
            self.__loramac_s = 'FFFFFFFFFFFFFFFF'

        adv_text = self.__advancedPageWidgets[1].currentText()
        if adv_text == INFO_TYPES[0]:
            if self.__res_json:
                self.__advancedPageWidgets[2].setText(self.get_display_name(self.__res_json['firmware_type']))
            else:
                self.__advancedPageWidgets[2].setText(self.__upgrade.get_chip_id())
        if adv_text == INFO_TYPES[1]:
            self.__advancedPageWidgets[2].setText(self.__w_mac_s)
        if adv_text == INFO_TYPES[2]:
            self.__advancedPageWidgets[2].setText(self.__loramac_s)
        if adv_text == INFO_TYPES[3]:
            self.__advancedPageWidgets[2].setText(sigfox_id)
        if adv_text == INFO_TYPES[4]:
            self.__advancedPageWidgets[2].setText(sigfox_pac)
        return

    def __newAdvancedVersionSelected(self):

        downgrade_details = self.__downgrade_versions.get(selected_version, None)

            self.__deviceInfoQuery.add_timeout(5)

            self.__findNewVersionMsg[5].setVisible(UPDATER_TYPE != 'internal')
            self.__findNewVersionMsg[0].setText('Fetching updates for this tool...')
            self.__findNewVersionMsg[1].show()
            wizard.button(QtGui.QWizard.NextButton).setEnabled(False)
            wizard.button(QtGui.QWizard.BackButton).setEnabled(False)
            wizard.button(QtGui.QWizard.CustomButton1).setEnabled(False)
        else:

    def __insert_callback(self, res):
        self.__deviceInsertAttempted = True
        self.__sigfox_registration_attempted = True
        print_debug('This is __insert_callback...')
        self.__forceAdvancedSettingsPage = True
        if res is not None:
            self.__res_status = res[0]
            if self.__res_status == 200:
                print_debug('self.__res_status==%s' % str(self.__res_status))
                self.__deviceInfoQuery.get_info(self.__w_mac_s, self.__loramac_s, self.__get_info_callback)
            else:
                print_debug('self.__res_status==%s' % str(self.__res_status))
                self.__server_connection_failed = True
                self.__finishedUpgrade(1, ('Code: [{}]').format(self.__res_status))
                try:
                    self.currentPage().jumpPage(self.RESULT_IDX)
                except Exception as e:
                    self.print_exception(e)
                    self.next()

        else:
            print_debug('res==None!')
            self.__server_connection_failed = True
            self.__finishedUpgrade(1)
            self.next()
        return

    def __read_device_config(self, res):
        if type(res) != bool and self.__upgrade is None or self.__upgrade.npy is None:
            self.__board_connection_failed = True
            self.__finishedUpgrade(1, str(res))
        else:
            try:
                print_debug('Trying to read the mac & config block...')
                w_mac = self.__upgrade.npy.read_mac()
                self.__w_mac_s = ''
                for n in w_mac:
                    self.__w_mac_s += ('{:02X}').format(n)

                self.__config_block = self.__upgrade.npy.read(int(PARTITIONS.get('config')[0], 16), int(PARTITIONS.get('config')[1], 16))
                self.__loramac_s = '%02X%02X%02X%02X%02X%02X%02X%02X' % struct.unpack('BBBBBBBB', self.__config_block[:8])
                if self.__loramac_s != 'FFFFFFFFFFFFFFFF' and self.__loramac_s != '0000000000000000':
                    self.__s_mac_s = [{'id': (self.__loramac_s), 'type': 'lpwan'}]
                else:
                    self.__s_mac_s = []
                try:
                    self.__wifi_on_boot = self.__config_block[88]
                except:
                    self.__wifi_on_boot = 1

                try:
                    self.__lora_region = ord(self.__config_block[52])
                except:
                    self.__lora_region = None

                if self.__lora_region == 255 or self.__forceNewLoRaRegionSelector.isChecked():
                    self.__lora_region = None
                else:
                    self.__lora_region = self.__upgrade.npy.region2str(self.__lora_region)
                try:
                    self.__wifi_ssid = self.__config_block[54:86].split('\x00')[0].decode('utf-8')
                except:
                    self.__wifi_ssid = ''

                try:
                    self.__wifi_pwd = self.__config_block[87:151].split('\x00')[0].decode('utf-8')
                except:
                    self.__wifi_pwd = ''

                try:
                    self.__pybytes_device_token = self.__config_block[162:201].split('\x00')[0].decode('utf-8')
                except:
                    self.__pybytes_device_token = None

                try:
                    self.__pybytes_mqttServiceAddress = self.__config_block[202:241].split('\x00')[0].decode('utf-8')
                except:
                    self.__pybytes_mqttServiceAddress = None

                try:
                    self.__pybytes_userId = self.__config_block[242:341].split('\x00')[0].decode('utf-8')
                except:
                    self.__pybytes_userId = None

                try:
                    self.__pybytes_nwprefs = self.__config_block[342:397].split('\x00')[0].decode('utf-8')
                except:
                    self.__pybytes_nwprefs = None

                try:
                    self.__pybytes_eprefs = self.__config_block[397:496].split('\x00')[0].decode('utf-8')
                except:
                    self.__pybytes_eprefs = None

                try:
                    self.__fstype = ord(self.__config_block[533])
                except:
                    self.__fstype = None

                try:
                    self.__pybytes_lte_carrier = self.__config_block[634:763].split('\x00')[0].decode('utf-8')
                except:
                    self.__pybytes_lte_carrier = None

                try:
                    self.__pybytes_lte_apn = self.__config_block[763:892].split('\x00')[0].decode('utf-8')

                    self.__pybytes_lte_band = ord(self.__config_block[910])
                except:
                    self.__pybytes_lte_band = None

                try:
                    self.__pybytes_lte_reset = ord(self.__config_block[911] == 1)
                except:
                    self.__pybytes_lte_reset = None

                print_debug(('wmac: {}').format(self.__w_mac_s))
                print_debug(('smac: {}').format(self.__loramac_s))
                print_debug(('region: {}').format(self.__lora_region))
                print_debug(('ssid: {}').format(self.__wifi_ssid))

                print_debug(('mqttServiceAddress: {}').format(self.__pybytes_mqttServiceAddress))
                print_debug(('userId: {}').format(self.__pybytes_userId))
                print_debug(('nwprefs: {}').format(self.__pybytes_nwprefs))
                print_debug(('eprefs: {}').format(self.__pybytes_eprefs))
                print_debug(('fstype: {}').format(self.__fstype))
                print_debug(('lte_carrier: {}').format(self.__pybytes_lte_carrier))
                print_debug(('lte_apn: {}').format(self.__pybytes_lte_apn))
                print_debug(('lte_type: {}').format(self.__pybytes_lte_type))
                print_debug(('lte_cid: {}').format(self.__pybytes_lte_cid))
                print_debug(('lte_band: {}').format(self.__pybytes_lte_band))
                print_debug(('lte_reset: {}').format(self.__pybytes_lte_reset))
                if self.__forceNewPybytesRegSelector.isChecked():
                    print_debug('Unsetting pybytes config block variables...')
                    self.__pybytes_device_token = None
                    self.__pybytes_mqttServiceAddress = None
                    self.__pybytes_userId = None
                    self.__pybytes_nwprefs = None
                    self.__pybytes_eprefs = None
                    self.__pybytes_lte_carrier = None
                    self.__pybytes_lte_apn = None
                    self.__pybytes_lte_type = None
                    self.__pybytes_lte_cid = None
                    self.__pybytes_lte_band = None
                    self.__pybytes_lte_reset = None
                if not self.__server_connection_failed:

                    except Exception as e:

                        self.__server_connection_failed = True

        try:

                self.__advancedPageWidgets[10].removeItem(0)
            freq = countryList[index - 2][1]

            self.__restoreAdvancedLoRaRegionIndex = self.__frequencySelectorLoPy.currentIndex()

            print_debug('<<<<< WARNING >>>>>> self.__lora_region is now %s' % self.__lora_region)
            if freq == 'EU868':
                self.__legacy_frequency = '868'
            else:
                self.__legacy_frequency = '915'

    def __fillPortList(self):
        try:
            try:
                import serial.tools.list_ports
                self.__portSelector.clear()
                topUsb = 0
                portList = []
                for n, (portname, desc, hwid) in enumerate(sorted(serial.tools.list_ports.comports(), reverse=True)):
                    if portname in SKIP_PORTS:
                        pass
                    elif check_usbid(portname):
                        print_debug('Port: %s Desc: %s HWID: %s' % (portname, desc, hwid))
                        portList.insert(topUsb, portname)
                        topUsb += 1
                    else:
                        portList.append(portname)

                self.__portSelector.addItems(portList)
                print_debug(('Trying to set self.__lastSerialPortUsed={}').format(self.__lastSerialPortUsed))
                try:
                    print_debug(('Do we have an index? {}').format(portList.index(self.__lastSerialPortUsed)))
                    self.__portSelector.setCurrentIndex(portList.index(self.__lastSerialPortUsed))
                except ValueError:
                    print_debug('Nope got ValueError')
                except Exception as e:
                    pass

            except Exception as e:
                self.print_exception(e)

        finally:
            self.__newSerialPortSelected()

        return

    def __upgradeBoard(self, res=None):
        if self.__forceNewLoRaRegionSelector.isChecked() and self.__localFileFlash:
            self.__lora_region = self.__frequencySelectorLoPy.currentText()
        if self.__res_json:
            remote_config_block = base64.b64decode(self.__res_json['binary'])
        print_debug('Setting config_block based on lora_region=%s' % self.__lora_region)
        if not self.__advancedPageWidgets[15].isChecked():
            config_block = self.__config_block
        else:
            config_block = self.__config_block[0:52].ljust(4096, b'\xff')
        config_block = self.__upgrade.npy.set_pycom_config(config_block, self.__check_update_fstype())
        config_block = self.__upgrade.npy.set_lpwan_config(config_block, str(self.__lora_region))
        if self.__pybytes_info_obtained:
            print_debug(('Updating pybytes_config: self.__enable_pybytes={}').format(self.__enable_pybytes))
            config_block = self.__upgrade.npy.set_pybytes_config(config_block, self.__pybytes_userId, self.__pybytes_device_token, self.__pybytes_mqttServiceAddress, self.__pybytes_nwprefs, self.__pybytes_eprefs, self.__forceupdate_pybytes, self.__enable_pybytes)
            config_block = self.__upgrade.npy.set_lte_config(config_block, self.__pybytes_lte_carrier, self.__pybytes_lte_apn, self.__pybytes_lte_type, self.__pybytes_lte_cid, self.__pybytes_lte_band, self.__pybytes_lte_reset)
        print_debug(('Updating pybytes_config: self.__enable_pybytes={}').format(self.__enable_pybytes))
        config_block = self.__upgrade.npy.set_pybytes_config(config_block, auto_start=self.__enable_pybytes)
        if self.__pybytes_info_obtained or self.__showAdvancedSettingsSelector.isChecked() and not self.__advancedPageWidgets[15].isChecked():
            print_debug(('!!!!!!!!! UPDATING WIFI CONFIG {}/{}').format(str(self.__wifi_ssid), str(self.__wifi_pwd)))
            config_block = self.__upgrade.npy.set_wifi_config(config_block, str(self.__wifi_ssid), str(self.__wifi_pwd))
        self.__upgrade.upgrade(self.__firmwareScript, self.__advancedPageWidgets[7].isChecked() if self.__showAdvancedSettingsSelector.isChecked() else self.__eraseFileSystemSelector.isChecked(), config_block, remote_config_block if self.__res_json else None, True, self.__progressLabel, self.__attemptLabel, self.__advancedPageWidgets[14].isChecked(), self.__finishedUpgrade)
        return

    def __check_advanced_page(self, original_return):

            print_debug('Returning CustomWizard.ADVANCED_IDX from __check_advanced_page')

            if self.__firmwareTypeSelector.currentText() in LICENSE_TYPES:
                self.__postlicense_page = original_return
                return CustomWizard.LICENSE_IDX
            return original_return

        return

    def __finishedUpgrade(self, res, error_msg=None):
        print_debug(('This is __finishedUpgrade({}, {})').format(res, error_msg))
        if type(res) is str:
            self.__error_message = res
            res = 1
        if error_msg is not None:
            self.__error_message = str(error_msg)
        if self.__error_message is not None:
            print_debug('ERROR MSG: ' + self.__error_message)
        self.__resultText.setWordWrap(True)
        self.__resultText.setOpenExternalLinks(True)
        success = False
        remote_msg = None
        if res == 0 or res > 10:
            self.__resultLabel.clear()
            self.__resultLabel.adjustSize()
            self.__resultLabel.setCenterOnScroll(False)
            self.__resultLabel.repaint()
            self.__resultLabel.setVisible(True)
            success = True
            print_debug(('self.__onlineFirmwareVersionTypes={}').format(self.__onlineFirmwareVersionTypes))
            if not self.__res_json:
                devname = 'device' if 1 else '<b>' + self.__onlineFirmwareVersionTypes[str(self.__firmwareTypeSelector.currentText())][self.__res_json['firmware_type']].get('prefix') + '</b>'
                final_text = 'Your ' + devname + ' was successfully updated.'
                if self.__res_json and not self.__localFileFlash:
                    fwtype = self.__onlineFirmwareVersionTypes[str(self.__firmwareTypeSelector.currentText())][self.__res_json['firmware_type']].get('type')
                    if fwtype == 'unstable':
                        fwtype = 'development'
                    if self.__flashAdvancedFirmwareVersion is not None:
                        fwversion = self.__flashAdvancedFirmwareVersion
                    else:
                        fwversion = self.__onlineFirmwareVersionTypes[str(self.__firmwareTypeSelector.currentText())][self.__res_json['firmware_type']].get('version')
                    final_text += '<br><br>Version: <b> ' + fwversion + '</b> [' + fwtype + ']'
                else:
                    final_text += '<br><br>File: <b> ' + self.__localFileName.split('/')[-1] + '</b>'
                if self.__start_time is not None:
                    final_text += '<br><br>Total time: <b>' + humanfriendly.format_timespan(time.time() - self.__start_time) + '</b>'
                if self.__res_json:
                    config_block = base64.b64decode(self.__res_json['binary'])
                else:
                    config_block = self.__config_block
                try:
                    sigfox_id = str(binascii.hexlify(config_block[8:12]).upper())
                    sigfox_pac = str(binascii.hexlify(config_block[12:20]).upper())
                except:
                    sigfox_id = 'FFFFFFFF'

                self.__pypic or self.__resultUpdateList.append('<br><br>Please remove the wire and <b>reset</b> the board.<br>')
            else:
                try:
                    self.__upgrade.npy.exit_pycom_programming_mode()
                except:
                    success = False
                    remote_msg = 'Could not exit programming mode'
                    self.__resultUpdateList.append(('<b>{}!</b><br><br>').format(remote_msg))

            self.__resultText.setText(final_text)
            for item in self.__resultUpdateList:
                self.__resultLabel.appendHtml(item + '<br>')

            if self.__w_mac_s is not None and len(self.__w_mac_s) >= 6:
                self.__resultLabel.appendHtml(('<b>Device ID</b>: {}').format(self.__w_mac_s.upper()))
                try:
                    self.__loramac_s = '%02X%02X%02X%02X%02X%02X%02X%02X' % struct.unpack('BBBBBBBB', config_block[:8])
                except:
                    self.__loramac_s = None

                if self.__loramac_s is not None and self.__loramac_s != 'FFFFFFFFFFFFFFFF' and self.__loramac_s != '0000000000000000':
                    self.__resultLabel.appendHtml(('<br><b>LoRa MAC</b>: {}').format(self.__loramac_s.upper()))
                if sigfox_id != 'FFFFFFFF':
                    self.__resultLabel.appendHtml('<br><b>Sigfox ID</b>: ' + sigfox_id + '<br>')
                    self.__resultLabel.appendHtml('<b>Sigfox PAC</b>: ' + sigfox_pac)
            new_stream.truncate(0)
        elif res == 1:
            self.__resultLabel.setVisible(False)
            if len(self.__resultUpdateList) > 0:
                for item in self.__resultUpdateList:
                    self.__resultLabel.appendHtml(item + '<br>')

                self.__resultLabel.moveCursor(QtGui.QTextCursor().atStart())
                self.__resultLabel.ensureCursorVisible()
                self.__resultLabel.setVisible(True)
                self.__resultLabel.adjustSize()
            if self.__w_mac_s is not None and len(self.__w_mac_s) >= 6:
                self.__resultLabel.appendHtml(('<b>Device ID </b>: {}').format(self.__w_mac_s.upper()))
                self.__resultLabel.moveCursor(QtGui.QTextCursor().atStart())
                self.__resultLabel.ensureCursorVisible()
                self.__resultLabel.setVisible(True)
                self.__resultLabel.adjustSize()
            if self.__upgrade or self.__board_connection_failed:
                if self.__error_message is not None:
                    if self.__error_message.startswith('Code'):
                        remote_msg = self.__error_message
                    else:
                        remote_msg = self.__error_message.split(']')[-1].replace("'", '')
                    final_msg = 'The upgrade failed due to the following error:<br><br>'
                    final_msg += '<b>' + remote_msg + '</b>'
                else:
                    remote_msg = 'Could not connect to the board'
                    final_msg = '<b>' + remote_msg + '.</b><br><br>'
                self.__resultText.setWordWrap(True)
                self.__resultText.setText(final_msg)
            elif self.__server_connection_failed:
                if self.__client_tool_oudated:
                    remote_msg = 'Client tool outdated'
                    self.__resultText.setText('<b>Your firmware update tool is too old!</b><br><br>Please download the latest version!\n                                                 <a href="' + SERVER_URL + '/findupgrade?strict=true&product=pycom-firmware-updater&type=' + self.__updater_type + '&platform=\n                                                 ' + detectOsFamily().lstrip() + '&redirect=true"> at this link </a><br><br>\n                                                 If you see this message after upgrading please email\n                                                 <a href="mailto:support@pycom.io">support@pycom.io')
                else:
                    if self.__error_message is not None:
                        remote_msg = self.__error_message
                    else:
                        remote_msg = 'Could not connect to the upgrade server!'
                    self.__resultText.setText('<b>' + remote_msg + '</b> <br><br>You can <b>Restart</b> to try again.<br><br>\n                                                 If the problem persists please email <a href="mailto:support@pycom.io">support@pycom.io</a><br><br>')
            else:
                remote_msg = 'Generic server-side error'
                self.__resultText.setText('<b>The upgrade failed due to a server side error.</b><br> Please try again in a few minutes. <br>\n                                             If the problem persists please email <a href="mailto:support@pycom.io">support@pycom.io</a><br><br>')
        elif res == 2:
            self.__resultLabel.setVisible(False)
            if len(self.__resultUpdateList) > 0:
                for item in self.__resultUpdateList:
                    self.__resultLabel.appendHtml(item + '<br>')

                self.__resultLabel.moveCursor(QtGui.QTextCursor().atStart())
                self.__resultLabel.ensureCursorVisible()
                self.__resultLabel.setVisible(True)
                self.__resultLabel.adjustSize()
            if self.__w_mac_s is not None and len(self.__w_mac_s) >= 6:
                self.__resultLabel.appendHtml(('<b>Device ID </b>: {}').format(self.__w_mac_s.upper()))
                self.__resultLabel.moveCursor(QtGui.QTextCursor().atStart())
                self.__resultLabel.ensureCursorVisible()
                self.__resultLabel.setVisible(True)
                self.__resultLabel.adjustSize()
            if self.__error_message is not None:
                remote_msg = self.__error_message.replace('"', '').replace("u'", '').replace("'", '')
            else:
                remote_msg = 'unknown error'
            self.__resultText.setText('The upgrade failed because the firmware file could not be read.<br><br><b>' + remote_msg + '</b>')
        try:
            if not self.__server_connection_failed and self.__consentSelector.isChecked():
                self.__deviceInfoQuery.send_update_status(self.__w_mac_s, success, remote_msg, self.__sendUpdateSatusCallback)
        except Exception as e:
            self.print_exception(e)

        self.next()
        return
